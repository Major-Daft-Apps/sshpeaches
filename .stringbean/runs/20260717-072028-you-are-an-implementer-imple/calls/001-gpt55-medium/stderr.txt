   813	        intent?.getStringExtra(EXTRA_START_ROUTE)
   814	            ?.takeIf(::isSupportedStartupRoute)

   820	                    UiDebugLog.result("handleSessionOpenIntent", false, "missing-host-id")
   821	                    return
   822	                }

   824	                requestedOpenSessionFileTransferEntryMode.value = null
   825	                UiDebugLog.result("handleSessionOpenIntent", true, "hostId=$hostId")
   826	            }
   827	
   828	            ACTION_WIDGET_CONNECT -> {
   829	                if (!HostWidgets.isTrustedWidgetActionIntent(this, intent)) {

   831	                    return
   832	                }

   836	                }.getOrDefault(ConnectionMode.SSH)
   837	                val fileTransferEntryMode = intent.getStringExtra(EXTRA_WIDGET_FILE_TRANSFER_ENTRY_MODE)
   838	                    ?.let { value ->
   839	                        runCatching { FileTransferEntryMode.valueOf(value) }.getOrNull()
   840	                    }

   265	        val prefs = context.getSharedPreferences(PREFS_NAME, Context.MODE_PRIVATE)
